# Independent surface-input and attitude-input AVANTI runtimes.
cmake_minimum_required(VERSION 3.16)
project(teso_diff_runtime VERSION 0.2.0 LANGUAGES CXX)
option(AVANTI_BUILD_FIXEDWING "Build original surface-input plant" ON)
option(AVANTI_BUILD_FIXEDWING_ATT "Build plant with generated attitude loop" ON)
option(AVANTI_BUILD_TESTS "Build native ABI validation executables" ON)
option(AVANTI_STAGE_LIBS "Copy shared libraries to runtime/*/lib" ON)
if(NOT AVANTI_BUILD_FIXEDWING AND NOT AVANTI_BUILD_FIXEDWING_ATT)
    message(FATAL_ERROR "Select at least one AVANTI runtime")
endif()
include(runtime/common/avanti_runtime.cmake)
if(AVANTI_BUILD_TESTS)
    enable_testing()
endif()
if(AVANTI_BUILD_FIXEDWING)
    add_subdirectory(runtime/fixedwing)
    if(AVANTI_BUILD_TESTS)
        avanti_add_test(avanti_cpp_20s tests/cpp/cpp_20s_test.cpp avanti2)
    endif()
endif()
if(AVANTI_BUILD_FIXEDWING_ATT)
    add_subdirectory(runtime/fixedwing_att)
    if(AVANTI_BUILD_TESTS)
        avanti_add_test(avanti_att_cpp_smoke tests/cpp/att_cpp_smoke_test.cpp avanti2_att)
    endif()
endif()
